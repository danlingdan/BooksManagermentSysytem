### 2.2 报表导出

private void ExportToExcel()
{
    using (var workbook = new XLWorkbook())
    {
        var worksheet = workbook.Worksheets.Add("报表");
        
        // 添加标题
        worksheet.Cell(1, 1).Value = "借阅统计报表";
        worksheet.Cell(2, 1).Value = $"统计周期：{startDate} ~ {endDate}";
        
        // 添加数据
        int row = 4;
        foreach (var data in reportData)
        {
            worksheet.Cell(row, 1).Value = data.Date;
            worksheet.Cell(row, 2).Value = data.BorrowCount;
            worksheet.Cell(row, 3).Value = data.ReturnCount;
            row++;
        }
        
        workbook.SaveAs("report.xlsx");
    }
}

## 3. 借阅统计报表

### 3.1 日借阅统计

```sql
SELECT 
    CAST(borrowdate AS DATE) AS 日期,
    COUNT(*) AS 借书次数,
    COUNT(DISTINCT cardID) AS 借书人数,
    SUM(CASE WHEN overdate IS NOT NULL THEN 1 ELSE 0 END) AS 还书次数
FROM borrow_record
WHERE borrowdate >= @startDate AND borrowdate < @endDate
GROUP BY CAST(borrowdate AS DATE)
ORDER BY 日期
```

### 3.2 月度借阅统计

```sql
SELECT 
    YEAR(borrowdate) AS 年,
    MONTH(borrowdate) AS 月,
    COUNT(*) AS 借书次数,
    COUNT(DISTINCT cardID) AS 借书人数,
    COUNT(DISTINCT bibliography_id) AS 借阅图书数
FROM borrow_record br
INNER JOIN bookborrow bb ON br.borrow_record_id = bb.borrow_record_id
WHERE YEAR(borrowdate) = @year AND MONTH(borrowdate) = @month
GROUP BY YEAR(borrowdate), MONTH(borrowdate)
```

### 3.3 图书借阅排行

```sql
-- 最借阅的图书（TOP 20）
SELECT TOP 20
    bib.bibliography_id,
    bib.bibliography_name,
    bib.category_id,
    COUNT(DISTINCT bb.bookborrow_id) AS 借阅次数,
    COUNT(DISTINCT bb.cardID) AS 借阅人数
FROM BIBLIOGRAPHY bib
INNER JOIN BOOK_ITEM bi ON bib.bibliography_id = bi.bibliography_id
INNER JOIN bookborrow bb ON bi.item_barcode = bb.bookID
WHERE bb.borrowdate >= @startDate AND bb.borrowdate < @endDate
GROUP BY bib.bibliography_id, bib.bibliography_name, bib.category_id
ORDER BY 借阅次数 DESC
```

### 3.4 读者借阅排行

```sql
-- 最活跃的读者（TOP 20）
SELECT TOP 20
    r.cardID,
    r.readername,
    r.readertype,
    COUNT(DISTINCT bb.bookborrow_id) AS 借阅次数,
    MAX(bb.borrowdate) AS 最后借阅日期
FROM reader r
INNER JOIN bookborrow bb ON r.cardID = bb.cardID
WHERE bb.borrowdate >= @startDate AND bb.borrowdate < @endDate
GROUP BY r.cardID, r.readername, r.readertype
ORDER BY 借阅次数 DESC
```

## 4. 读者统计报表

### 4.1 读者总量统计

```sql
SELECT 
    COUNT(*) AS 总读者数,
    SUM(CASE WHEN readertype=N'本校学生' THEN 1 ELSE 0 END) AS 学生数,
    SUM(CASE WHEN readertype=N'本校教师' THEN 1 ELSE 0 END) AS 教师数,
    SUM(CASE WHEN readertype=N'校外人员' THEN 1 ELSE 0 END) AS 校外数
FROM reader
```

### 4.2 读者注册趋势

```sql
SELECT 
    CAST(rc.startdate AS DATE) AS 日期,
    COUNT(*) AS 新增人数,
    SUM(CASE WHEN r.readertype=N'本校学生' THEN 1 ELSE 0 END) AS 学生,
    SUM(CASE WHEN r.readertype=N'本校教师' THEN 1 ELSE 0 END) AS 教师
FROM readcard rc
INNER JOIN reader r ON rc.cardID = r.cardID
WHERE rc.startdate >= @startDate AND rc.startdate < @endDate
GROUP BY CAST(rc.startdate AS DATE)
ORDER BY 日期
```

### 4.3 借书证状态分布

```sql
SELECT 
    rc.state AS 状态,
    COUNT(*) AS 数量,
    CAST(COUNT(*)*100.0/(SELECT COUNT(*) FROM readcard) AS DECIMAL(5,2)) AS 占比
FROM readcard rc
GROUP BY rc.state
```

## 5. 图书统计报表

### 5.1 馆藏统计

```sql
SELECT 
    COUNT(DISTINCT bi.bibliography_id) AS 书目数,
    COUNT(*) AS 馆藏数,
    SUM(CASE WHEN bi.current_status='AVAILABLE' THEN 1 ELSE 0 END) AS 可借数,
    SUM(CASE WHEN bi.current_status='BORROWED' THEN 1 ELSE 0 END) AS 已借数,
    SUM(CASE WHEN bi.current_status='OFF_SHELF' THEN 1 ELSE 0 END) AS 下架数
FROM BOOK_ITEM bi
```

### 5.2 图书分类分布

```sql
SELECT TOP 20
    bc.category_code,
    bc.category_name,
    COUNT(DISTINCT bib.bibliography_id) AS 书目数,
    COUNT(*) AS 馆藏数,
    ROUND(CAST(COUNT(*)*100.0 / 
        (SELECT COUNT(*) FROM BOOK_ITEM) AS DECIMAL(5,2)), 2) AS 占比
FROM BOOK_CATEGORY bc
LEFT JOIN BIBLIOGRAPHY bib ON bc.category_id = bib.category_id
LEFT JOIN BOOK_ITEM bi ON bib.bibliography_id = bi.bibliography_id
GROUP BY bc.category_id, bc.category_code, bc.category_name
ORDER BY 馆藏数 DESC
```

### 5.3 库位容量统计

```sql
SELECT 
    location_code,
    location_name,
    max_capacity,
    current_quantity,
    max_capacity - current_quantity AS 剩余容量,
    ROUND(CAST(current_quantity*100.0/max_capacity AS DECIMAL(5,2)), 2) AS 占用率
FROM STORAGE_LOCATION
WHERE status = 'ACTIVE'
ORDER BY 占用率 DESC
```

## 6. 财务统计报表

### 6.1 罚款统计

```sql
SELECT 
    SUM(amount) AS 总罚款,
    SUM(CASE WHEN fine_status=N'已支付' THEN amount ELSE 0 END) AS 已收取,
    SUM(CASE WHEN fine_status=N'未支付' THEN amount ELSE 0 END) AS 未收取,
    COUNT(*) AS 罚款笔数,
    AVG(amount) AS 平均罚款
FROM fine
WHERE created_time >= @startDate AND created_time < @endDate
```

### 6.2 罚款原因分布

```sql
SELECT 
    reason AS 罚款原因,
    COUNT(*) AS 次数,
    SUM(amount) AS 总金额,
    AVG(amount) AS 平均金额
FROM fine
WHERE created_time >= @startDate AND created_time < @endDate
GROUP BY reason
ORDER BY 总金额 DESC
```

### 6.3 逾期统计

```sql
SELECT 
    COUNT(*) AS 逾期次数,
    COUNT(DISTINCT cardID) AS 逾期人数,
    AVG(DATEDIFF(DAY, overdate, GETDATE())) AS 平均逾期天数,
    MAX(DATEDIFF(DAY, overdate, GETDATE())) AS 最多逾期天数
FROM bookborrow
WHERE overdate IS NULL AND 
      borrowdate + (SELECT borrow_days FROM BORROW_RULES WHERE reader_type='本校学生') 
      < GETDATE()
```

### 8.3 定时报表

```csharp
// 每日自动生成报表
public void ScheduleDailyReport()
{
    Timer timer = new Timer();
    timer.Interval = 24 * 60 * 60 * 1000; // 24小时
    timer.Elapsed += (s, e) =>
    {
        // 生成日报表
        var report = GenerateDailyReport(DateTime.Today);
        
        // 保存到文件
        SaveReport(report);
        
        // 发送邮件通知
        SendReportEmail(report);
    };
    timer.Start();
}
```

## 9. 数据分析

### 9.1 热力图分析

显示借阅热度分布：

```sql
-- 按分类和时间的热力分析
SELECT 
    bc.category_name,
    DATEPART(HOUR, bb.borrowdate) AS 小时,
    COUNT(*) AS 借阅次数
FROM bookborrow bb
INNER JOIN BOOK_ITEM bi ON bb.bookID = bi.item_barcode
INNER JOIN BIBLIOGRAPHY bib ON bi.bibliography_id = bib.bibliography_id
INNER JOIN BOOK_CATEGORY bc ON bib.category_id = bc.category_id
GROUP BY bc.category_name, DATEPART(HOUR, bb.borrowdate)
ORDER BY bc.category_name, DATEPART(HOUR, bb.borrowdate)
```

### 9.2 趋势分析

```sql
-- 借阅趋势
SELECT 
    DATEADD(DAY, -DATEPART(DAY, borrowdate)+1, borrowdate) AS 月份,
    COUNT(*) AS 借书次数,
    COUNT(DISTINCT cardID) AS 借书人数
FROM borrow_record
GROUP BY DATEADD(DAY, -DATEPART(DAY, borrowdate)+1, borrowdate)
ORDER BY 月份
```
