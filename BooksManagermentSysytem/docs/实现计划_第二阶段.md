```sql
-- 借阅综合查询视图
CREATE VIEW dbo.vw_borrow_comprehensive AS
SELECT 
    r.cardID,
    r.readername,
    r.readertype,
    r.unit,
    r.number AS student_employee_no,
    bb.bookborrow_id,
    bb.bookID,
    bib.bibliography_name AS book_name,
    bib.ISBN,
    bc.category_code,
    bc.category_name,
    bb.borrowdate,
    DATEADD(DAY, 7, bb.borrowdate) AS due_date,
    bb.overdate AS return_date,
    CASE 
        WHEN bb.overdate IS NULL AND GETDATE() > DATEADD(DAY, 7, bb.borrowdate) 
        THEN DATEDIFF(DAY, DATEADD(DAY, 7, bb.borrowdate), GETDATE())
        WHEN bb.overdate IS NOT NULL AND bb.overdate > DATEADD(DAY, 7, bb.borrowdate)
        THEN DATEDIFF(DAY, DATEADD(DAY, 7, bb.borrowdate), bb.overdate)
        ELSE 0 
    END AS overdue_days,
    CASE 
        WHEN bb.overdue IS NULL THEN N'在借'
        ELSE N'已还'
    END AS borrow_status,
    br.bcomplete AS book_condition
FROM dbo.bookborrow bb
INNER JOIN dbo.reader r ON bb.cardID = r.cardID
INNER JOIN dbo.BOOK_ITEM bi ON bb.bookID = bi.item_barcode
INNER JOIN dbo.BIBLIOGRAPHY bib ON bi.bibliography_id = bib.bibliography_id
INNER JOIN dbo.BOOK_CATEGORY bc ON bib.category_id = bc.category_id
LEFT JOIN dbo.borrow_record br ON bb.borrow_record_id = br.borrow_record_id;

-- 超期图书汇总视图
CREATE VIEW dbo.vw_overdue_books_summary AS
SELECT 
    r.unit AS department,
    COUNT(*) AS overdue_count,
    SUM(DATEDIFF(DAY, DATEADD(DAY, 7, bb.borrowdate), GETDATE())) AS total_overdue_days,
    STRING_AGG(r.readername + ' - ' + bib.bibliography_name, '; ') AS overdue_details
FROM dbo.bookborrow bb
INNER JOIN dbo.reader r ON bb.cardID = r.cardID
INNER JOIN dbo.BOOK_ITEM bi ON bb.bookID = bi.item_barcode
INNER JOIN dbo.BIBLIOGRAPHY bib ON bi.bibliography_id = bib.bibliography_id
WHERE bb.overdate IS NULL 
  AND DATEADD(DAY, 7, bb.borrowdate) < GETDATE()
GROUP BY r.unit;

-- 罚款明细汇总视图
CREATE VIEW dbo.vw_fine_summary AS
SELECT 
    f.cardID,
    r.readername,
    r.readertype,
    r.unit,
    COUNT(*) AS fine_count,
    SUM(CASE WHEN f.fine_status = N'未支付' THEN f.amount ELSE 0 END) AS unpaid_amount,
    SUM(CASE WHEN f.fine_status = N'已支付' THEN f.amount ELSE 0 END) AS paid_amount,
    SUM(f.amount) AS total_amount
FROM dbo.fine f
INNER JOIN dbo.reader r ON f.cardID = r.cardID
GROUP BY f.cardID, r.readername, r.readertype, r.unit;

-- 图书被借阅史视图
CREATE VIEW dbo.vw_book_borrow_history AS
SELECT 
    bi.item_barcode,
    bib.bibliography_name,
    bib.ISBN,
    COUNT(*) AS borrow_times,
    MIN(bb.borrowdate) AS first_borrow_date,
    MAX(bb.borrowdate) AS last_borrow_date,
    COUNT(CASE WHEN bb.overdate IS NULL THEN 1 END) AS current_borrowed,
    COUNT(CASE WHEN bb.overdate > DATEADD(DAY, 7, bb.borrowdate) THEN 1 END) AS overdue_times
FROM dbo.bookborrow bb
INNER JOIN dbo.BOOK_ITEM bi ON bb.bookID = bi.item_barcode
INNER JOIN dbo.BIBLIOGRAPHY bib ON bi.bibliography_id = bib.bibliography_id
GROUP BY bi.item_barcode, bib.bibliography_name, bib.ISBN;

-- 已注销/损坏图书查询视图
CREATE VIEW dbo.vw_inactive_damaged_books AS
SELECT 
    bi.item_barcode,
    bib.bibliography_name,
    bib.ISBN,
    bc.category_code,
    bc.category_name,
    bi.current_status,
    bi.physical_condition,
    bi.status_changed_date,
    sl.location_name
FROM dbo.BOOK_ITEM bi
INNER JOIN dbo.BIBLIOGRAPHY bib ON bi.bibliography_id = bib.bibliography_id
INNER JOIN dbo.BOOK_CATEGORY bc ON bib.category_id = bc.category_id
INNER JOIN dbo.STORAGE_LOCATION sl ON bi.location_id = sl.location_id
WHERE bi.current_status = N'OFF_SHELF' 
   OR bi.physical_condition IN (N'DAMAGED', N'REPAIR');
```

## Week 9-10: 流通统计与馆藏统计

### 1. 数据库存储过程

**流通统计存储过程：**
```sql
CREATE PROCEDURE dbo.sp_circulation_statistics
    @start_date DATE,
    @end_date DATE,
    @period_type NVARCHAR(10) = N'DAY' -- DAY/WEEK/MONTH/YEAR
AS
BEGIN
    -- 各时期图书流通率
    SELECT 
        CASE @period_type
            WHEN N'DAY' THEN CONVERT(VARCHAR(10), bb.borrowdate, 120)
            WHEN N'WEEK' THEN CONCAT(YEAR(bb.borrowdate), '-W', DATEPART(WEEK, bb.borrowdate))
            WHEN N'MONTH' THEN CONCAT(YEAR(bb.borrowdate), '-', FORMAT(MONTH(bb.borrowdate), '00'))
            WHEN N'YEAR' THEN CAST(YEAR(bb.borrowdate) AS VARCHAR(4))
        END AS period,
        COUNT(*) AS borrow_count,
        COUNT(DISTINCT bb.cardID) AS reader_count,
        COUNT(DISTINCT bb.bookID) AS book_count,
        CAST(COUNT(*) AS FLOAT) / NULLIF(
            (SELECT COUNT(*) FROM dbo.BOOK_ITEM WHERE current_status = N'AVAILABLE'), 0
        ) AS circulation_rate
    FROM dbo.bookborrow bb
    WHERE bb.borrowdate >= @start_date AND bb.borrowdate < DATEADD(DAY, 1, @end_date)
    GROUP BY 
        CASE @period_type
            WHEN N'DAY' THEN CONVERT(VARCHAR(10), bb.borrowdate, 120)
            WHEN N'WEEK' THEN CONCAT(YEAR(bb.borrowdate), '-W', DATEPART(WEEK, bb.borrowdate))
            WHEN N'MONTH' THEN CONCAT(YEAR(bb.borrowdate), '-', FORMAT(MONTH(bb.borrowdate), '00'))
            WHEN N'YEAR' THEN CAST(YEAR(bb.borrowdate) AS VARCHAR(4))
        END
    ORDER BY period;
END;

-- 分类流通率统计
CREATE PROCEDURE dbo.sp_category_circulation
    @start_date DATE,
    @end_date DATE
AS
BEGIN
    SELECT 
        bc.category_code,
        bc.category_name,
        COUNT(*) AS borrow_count,
        COUNT(DISTINCT bb.bookID) AS distinct_books,
        CAST(COUNT(*) AS FLOAT) / NULLIF(
            (SELECT COUNT(*) FROM dbo.BOOK_ITEM bi2
             INNER JOIN dbo.BIBLIOGRAPHY bib2 ON bi2.bibliography_id = bib2.bibliography_id
             WHERE bib2.category_id = bc.category_id), 0
        ) AS circulation_rate
    FROM dbo.bookborrow bb
    INNER JOIN dbo.BOOK_ITEM bi ON bb.bookID = bi.item_barcode
    INNER JOIN dbo.BIBLIOGRAPHY bib ON bi.bibliography_id = bib.bibliography_id
    INNER JOIN dbo.BOOK_CATEGORY bc ON bib.category_id = bc.category_id
    WHERE bb.borrowdate >= @start_date AND bb.borrowdate < DATEADD(DAY, 1, @end_date)
    GROUP BY bc.category_id, bc.category_code, bc.category_name
    ORDER BY borrow_count DESC;
END;

-- 馆藏统计存储过程
CREATE PROCEDURE dbo.sp_collection_statistics
    @period_type NVARCHAR(10) = N'MONTH'
AS
BEGIN
    -- 各时期馆藏量统计（基于入库日期）
    SELECT 
        CASE @period_type
            WHEN N'MONTH' THEN CONCAT(YEAR(bi.acquisition_date), '-', FORMAT(MONTH(bi.acquisition_date), '00'))
            WHEN N'YEAR' THEN CAST(YEAR(bi.acquisition_date) AS VARCHAR(4))
        END AS period,
        COUNT(*) AS new_books,
        SUM(COUNT(*)) OVER (ORDER BY 
            CASE @period_type
                WHEN N'MONTH' THEN CONCAT(YEAR(bi.acquisition_date), '-', FORMAT(MONTH(bi.acquisition_date), '00'))
                WHEN N'YEAR' THEN CAST(YEAR(bi.acquisition_date) AS VARCHAR(4))
            END
        ) AS cumulative_total
    FROM dbo.BOOK_ITEM bi
    GROUP BY 
        CASE @period_type
            WHEN N'MONTH' THEN CONCAT(YEAR(bi.acquisition_date), '-', FORMAT(MONTH(bi.acquisition_date), '00'))
            WHEN N'YEAR' THEN CAST(YEAR(bi.acquisition_date) AS VARCHAR(4))
        END
    ORDER BY period;
END;

-- 分类馆藏量统计
CREATE PROCEDURE dbo.sp_category_collection
AS
BEGIN
    SELECT 
        bc.category_code,
        bc.category_name,
        COUNT(*) AS total_books,
        COUNT(CASE WHEN bi.current_status = N'AVAILABLE' THEN 1 END) AS available_books,
        COUNT(CASE WHEN bi.current_status = N'BORROWED' THEN 1 END) AS borrowed_books,
        COUNT(CASE WHEN bi.physical_condition = N'DAMAGED' THEN 1 END) AS damaged_books,
        CAST(COUNT(CASE WHEN bi.current_status = N'BORROWED' THEN 1 END) AS FLOAT) / 
            NULLIF(COUNT(*), 0) AS utilization_rate
    FROM dbo.BOOK_ITEM bi
    INNER JOIN dbo.BIBLIOGRAPHY bib ON bi.bibliography_id = bib.bibliography_id
    INNER JOIN dbo.BOOK_CATEGORY bc ON bib.category_id = bc.category_id
    GROUP BY bc.category_id, bc.category_code, bc.category_name
    ORDER BY total_books DESC;
END;
```

```csharp
// 文件：BooksManagermentSysytem\src\Services\ExportService.cs
using ClosedXML.Excel;

public class ExportService
{
    public void ExportToExcel(DataTable data, string filePath, string sheetName)
    {
        using (var workbook = new XLWorkbook())
        {
            var worksheet = workbook.Worksheets.Add(sheetName);
            worksheet.Cell(1, 1).InsertTable(data);
            
            // 自动调整列宽
            worksheet.Columns().AdjustToContents();
            
            // 标题行格式化
            var headerRow = worksheet.Row(1);
            headerRow.Style.Font.Bold = true;
            headerRow.Style.Fill.BackgroundColor = XLColor.LightBlue;
            
            workbook.SaveAs(filePath);
        }
    }
    
    public void ExportStatisticsChart(ChartData data, string filePath)
    {
        // 导出带图表的 Excel
        using (var workbook = new XLWorkbook())
        {
            var worksheet = workbook.Worksheets.Add("统计分析");
            
            // 插入数据
            var table = worksheet.Cell(1, 1).InsertTable(data.Table);
            
            // 创建图表（ClosedXML 0.95+ 支持）
            var chart = worksheet.Charts.Add(10, 5, 30, 15);
            chart.SetChartType(XLChartType.Column);
            chart.SetDataRange(table.AsRange());
            
            workbook.SaveAs(filePath);
        }
    }
}
```

## 性能优化建议

1. **查询优化**
   - 为常用查询字段添加索引
   - 使用视图缓存复杂计算

2. **分页查询**
   - 使用 `OFFSET-FETCH` 或 `ROW_NUMBER()`
   - 限制单页记录数（建议 50-100 条）

3. **导出优化**
   - 大数据量分批导出
   - 使用流式写入避免内存溢出
